// COMPLETE WORKFLOW EXAMPLE
// =====================================================

async function completeOrderWorkflow() {
  // 1. Create order
  const order = await createOrder({
    order_type: 'dine_in',
    table_id: 'table-5-uuid',
    customer_name: 'Rajesh Kumar'
  });

  const orderId = order.id;

  // 2. Add items
  await addItems(orderId, [
    { product_id: 'momo-uuid', quantity: 2 },
    { product_id: 'chowmein-uuid', quantity: 1 },
    { product_id: 'coke-uuid', quantity: 2 }
  ]);

  // 3. Confirm order
  await updateStatus(orderId, 'confirmed');

  // 4. Kitchen updates
  await updateStatus(orderId, 'preparing');
  await updateStatus(orderId, 'ready');

  // 5. Serve food
  await updateStatus(orderId, 'served');

  // 6. Apply discount (optional)
  await applyDiscount(orderId, {
    discount_type: 'percentage',
    discount_value: 10
  });

  // 7. Split payment
  await addPayment(orderId, { method: 'cash', amount: 400 });
  await addPayment(orderId, { method: 'esewa', amount: 300 });
  await addPayment(orderId, { method: 'khalti', amount: 87.20 });

  // 8. Complete order
  await updateStatus(orderId, 'completed');

  console.log('Order completed successfully!');
}
