1️⃣ What is User-Agent (simple)

User-Agent is a string sent by the browser/app that tells the server:

Browser name & version

OS

Device type

Example:

Mozilla/5.0 (Windows NT 10.0; Win64; x64)
Chrome/121.0.0.0

In Express:

const userAgent = req.headers["user-agent"];


📌 Used for: detecting new devices, security alerts, analytics
📌 Not unique, but useful when combined with IP

// otp.service.ts
import crypto from "crypto";

export function generateOTP() {
  return Math.floor(100000 + Math.random() * 900000).toString(); // 6-digit
}

export function hashOTP(otp: string) {
  return crypto.createHash("sha256").update(otp).digest("hex");
}


import { Request, Response } from "express";
import { generateOTP, hashOTP } from "./otp.service";

let mockUser = {
  email: "test@example.com",
  password: "password123",
  last_login_ip: null as string | null,
  last_user_agent: null as string | null,
  otp_hash: null as string | null,
  otp_expires_at: null as Date | null,

export const login = (req: Request, res: Response) => {
  const { email, password } = req.body;

  const ip = req.ip;
  const userAgent = req.headers["user-agent"];

  // 1. Check credentials (mock)
  if (email !== mockUser.email || password !== mockUser.password) {
    return res.status(401).json({ message: "Invalid credentials" });
  }

  // 2. Detect new device
  const isNewDevice =
    mockUser.last_login_ip !== ip ||
    mockUser.last_user_agent !== userAgent;

  if (isNewDevice) {
    const otp = generateOTP();

    mockUser.otp_hash = hashOTP(otp);
    mockUser.otp_expires_at = new Date(Date.now() + 5 * 60 * 1000); // 5 min

    console.log("🔐 OTP sent (mock):", otp);

    return res.json({
      message: "New device detected. OTP required.",
      otpRequired: true,
    });
  }

  // 3. Normal login
  mockUser.last_login_ip = ip!;
  mockUser.last_user_agent = userAgent!;

  return res.json({ message: "Login successful" });
};
